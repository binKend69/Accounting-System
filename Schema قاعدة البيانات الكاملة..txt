القسم 1: الإعدادات الأساسية والمرجعية (Setup/Reference Tables)
-- ==========================================
-- 1. العملات
-- ==========================================
CREATE TABLE Currencies (
    CurrencyId          INT IDENTITY(1,1) PRIMARY KEY,
    NameAr               NVARCHAR(50)  NOT NULL,           -- ريال سعودي
    Symbol               NVARCHAR(10)  NULL,                -- ر.س
    FractionName         NVARCHAR(30)  NULL,                -- فلس / هللة
    ExchangeRate         DECIMAL(18,6) NOT NULL DEFAULT 1,  -- التعادل
    ConversionType       TINYINT       NOT NULL DEFAULT 1,  -- 1=ضرب, 2=قسمة
    IsBaseCurrency       BIT           NOT NULL DEFAULT 0,  -- العملة الأساسية للنظام (لعرض التقارير الافتراضية)
    IsActive             BIT           NOT NULL DEFAULT 1,
    CreatedAt            DATETIME2     NOT NULL DEFAULT SYSDATETIME(),
    ModifiedAt           DATETIME2     NULL,
    IsDeleted            BIT           NOT NULL DEFAULT 0   -- Soft Delete لدعم "سلة المحذوفات"
);

-- ==========================================
-- 2. الفروع
-- ==========================================
CREATE TABLE Branches (
    BranchId             INT IDENTITY(1,1) PRIMARY KEY,
    NameAr               NVARCHAR(100) NOT NULL,
    IsActive             BIT NOT NULL DEFAULT 1,
    CreatedAt            DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    ModifiedAt           DATETIME2 NULL,
    IsDeleted            BIT NOT NULL DEFAULT 0
);

-- ==========================================
-- 3. المراكز (مركز تكلفة/ربحية - Cost Centers)
-- ==========================================
CREATE TABLE Centers (
    CenterId             INT IDENTITY(1,1) PRIMARY KEY,
    NameAr               NVARCHAR(100) NOT NULL,
    BranchId             INT NULL REFERENCES Branches(BranchId),
    IsActive             BIT NOT NULL DEFAULT 1,
    CreatedAt            DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    ModifiedAt           DATETIME2 NULL,
    IsDeleted            BIT NOT NULL DEFAULT 0
);

-- ==========================================
-- 4. المستودعات
-- ==========================================
CREATE TABLE Warehouses (
    WarehouseId          INT IDENTITY(1,1) PRIMARY KEY,
    NameAr               NVARCHAR(100) NOT NULL,
    BranchId             INT NULL REFERENCES Branches(BranchId),
    IsActive             BIT NOT NULL DEFAULT 1,
    CreatedAt            DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    ModifiedAt           DATETIME2 NULL,
    IsDeleted            BIT NOT NULL DEFAULT 0
);

-- ==========================================
-- 5. الوحدات (وحدة أساسية عامة: حبة، كرتون، كيلو...)
-- ==========================================
CREATE TABLE Units (
    UnitId               INT IDENTITY(1,1) PRIMARY KEY,
    NameAr               NVARCHAR(50) NOT NULL,
    IsActive             BIT NOT NULL DEFAULT 1,
    CreatedAt            DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    ModifiedAt           DATETIME2 NULL,
    IsDeleted            BIT NOT NULL DEFAULT 0
);

-- ==========================================
-- 6. الأقسام (تصنيف الأصناف)
-- ==========================================
CREATE TABLE Departments (
    DepartmentId         INT IDENTITY(1,1) PRIMARY KEY,
    NameAr               NVARCHAR(100) NOT NULL,
    ColorHex             NVARCHAR(9)  NULL,          -- لون زر القسم بشاشة الكاشير (مثل #FFCC00)
    DisplayOrder         INT NOT NULL DEFAULT 0,     -- توحيد تسلسل الأقسام بالكاشير
    IsActive             BIT NOT NULL DEFAULT 1,
    CreatedAt            DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    ModifiedAt           DATETIME2 NULL,
    IsDeleted            BIT NOT NULL DEFAULT 0
);

-- ==========================================
-- 7. المواقع (موقع تخزين داخل المستودع - Bin Location)
-- ==========================================
CREATE TABLE StockLocations (
    LocationId           INT IDENTITY(1,1) PRIMARY KEY,
    NameAr               NVARCHAR(100) NOT NULL,
    WarehouseId          INT NULL REFERENCES Warehouses(WarehouseId),
    IsActive             BIT NOT NULL DEFAULT 1,
    CreatedAt            DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    ModifiedAt           DATETIME2 NULL,
    IsDeleted            BIT NOT NULL DEFAULT 0
);

-- ==========================================
-- 8. معلومات الشركة/الفاتورة (سطر واحد فقط - Singleton)
-- ==========================================
CREATE TABLE CompanyInfo (
    CompanyInfoId        INT IDENTITY(1,1) PRIMARY KEY,
    NameAr               NVARCHAR(200) NOT NULL,
    NameEn               NVARCHAR(200) NULL,
    AddressAr            NVARCHAR(300) NULL,
    AddressEn            NVARCHAR(300) NULL,
    Phone                NVARCHAR(30)  NULL,
    Mobile               NVARCHAR(30)  NULL,
    LogoPath             NVARCHAR(300) NULL,
    StampPath            NVARCHAR(300) NULL,          -- الختم
    UseImageInsteadOfText BIT NOT NULL DEFAULT 0,
    InvoiceFooterNote    NVARCHAR(500) NULL,           -- نص أسفل الفاتورة
    BaseCurrencyId       INT NULL REFERENCES Currencies(CurrencyId),
    -- إعدادات عامة إضافية:
    CostingMethod        TINYINT NOT NULL DEFAULT 1,   -- 1= آخر سعر شراء, 2= متوسط التكلفة
    DefaultPurchaseCurrencyId INT NULL REFERENCES Currencies(CurrencyId),
    DefaultSalesCurrencyId    INT NULL REFERENCES Currencies(CurrencyId),
    DefaultVoucherCurrencyId  INT NULL REFERENCES Currencies(CurrencyId),
    CreatedAt            DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    ModifiedAt           DATETIME2 NULL
);

ممتاز، تم تثبيت كل القرارات. الآن أبدأ ببناء **Schema قاعدة البيانات الكاملة**.

نظراً لضخامة النظام (محاسبة + مخزون + مبيعات/مشتريات + كاشير + رواتب + صلاحيات + تدقيق)، سأبنيه بأقسام منطقية متسلسلة حتى يسهل عليك المراجعة والتعقيب على كل قسم قبل الانتقال للتالي.

القسم 2: شجرة الحسابات والقيود المحاسبية (Chart of Accounts & Journal)
-- ==========================================
-- 1. المجموعات العامة (تصنيف موازٍ: عملاء/موردون/صناديق/بنوك...)
-- ==========================================
CREATE TABLE AccountGeneralGroups (
    GeneralGroupId       INT IDENTITY(1,1) PRIMARY KEY,
    NameAr               NVARCHAR(100) NOT NULL,    -- الجميع، أصول ثابتة، صناديق، بنوك، عملاء، موردون...
    SystemCode           NVARCHAR(30)  NULL,        -- كود ثابت داخلي (CASH, BANK, CUSTOMER, SUPPLIER...) يُستخدم بالكود لتمييزها برمجياً
    IsActive             BIT NOT NULL DEFAULT 1,
    IsDeleted            BIT NOT NULL DEFAULT 0
);

-- ==========================================
-- 2. المجموعات الرئيسية (شجرة الترقيم العشري الهرمي)
-- ==========================================
CREATE TABLE AccountMainGroups (
    MainGroupId          INT IDENTITY(1,1) PRIMARY KEY,
    Code                 NVARCHAR(10)  NOT NULL UNIQUE,   -- '1','11','111','1211'...
    NameAr               NVARCHAR(150) NOT NULL,
    ParentMainGroupId    INT NULL REFERENCES AccountMainGroups(MainGroupId), -- هرمية شجرية حقيقية
    ClosingReportType    TINYINT NULL,   -- 1=المتاجرة, 2=الأرباح والخسائر, 3=الميزانية العمومية, NULL=لا يظهر بحساب ختامي
    Level                INT NOT NULL DEFAULT 1,          -- عمق الشجرة (1، 2، 3...) لتسهيل الفرز والعرض
    IsActive             BIT NOT NULL DEFAULT 1,
    IsDeleted            BIT NOT NULL DEFAULT 0
);

-- ==========================================
-- 3. شجرة الحسابات (الحسابات التفصيلية الفعلية)
-- ==========================================
CREATE TABLE Accounts (
    AccountId            INT IDENTITY(1,1) PRIMARY KEY,
    AccountCode          NVARCHAR(20)  NOT NULL UNIQUE,  -- رقم الحساب (2111 مثلاً)
    NameAr               NVARCHAR(200) NOT NULL,
    MainGroupId          INT NOT NULL REFERENCES AccountMainGroups(MainGroupId),
    GeneralGroupId       INT NULL REFERENCES AccountGeneralGroups(GeneralGroupId),
    Nature               TINYINT NOT NULL,        -- 1=مدين, 2=دائن (طبيعة الحساب)
    BranchId             INT NULL REFERENCES Branches(BranchId),   -- خاص بفرع (NULL = عام لكل الفروع)
    CurrencyId           INT NOT NULL REFERENCES Currencies(CurrencyId),
    -- بيانات اتصال (للعملاء/الموردين):
    Phone                NVARCHAR(30)  NULL,
    Mobile               NVARCHAR(30)  NULL,
    Address              NVARCHAR(300) NULL,
    Notes                NVARCHAR(500) NULL,
    -- ضوابط:
    CreditLimit          DECIMAL(18,2) NULL,       -- الحد الأعلى للدين (الحد الأعلى مسموح به)
    OpeningBalance       DECIMAL(18,2) NOT NULL DEFAULT 0,
    OpeningBalanceNature TINYINT NULL,             -- 1=مدين, 2=دائن (لحالة الرصيد الافتتاحي)
    IsHidden             BIT NOT NULL DEFAULT 0,   -- إخفاء الحساب
    IsSuspended          BIT NOT NULL DEFAULT 0,   -- إيقاف الحساب (منع الحركة عليه)
    IsSystemAccount      BIT NOT NULL DEFAULT 0,   -- حساب نظامي (لا يُحذف، مثل "الصندوق العام")
    CreatedAt            DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    ModifiedAt           DATETIME2 NULL,
    IsDeleted            BIT NOT NULL DEFAULT 0
);

CREATE INDEX IX_Accounts_MainGroup ON Accounts(MainGroupId);
CREATE INDEX IX_Accounts_GeneralGroup ON Accounts(GeneralGroupId);

-- ==========================================
-- 4. السنوات/الفترات المالية (لدعم "الإقفال السنوي")
-- ==========================================
CREATE TABLE FiscalPeriods (
    FiscalPeriodId       INT IDENTITY(1,1) PRIMARY KEY,
    NameAr               NVARCHAR(50) NOT NULL,     -- "سنة 2026"
    StartDate            DATE NOT NULL,
    EndDate              DATE NOT NULL,
    IsClosed             BIT NOT NULL DEFAULT 0,
    ClosedAt             DATETIME2 NULL,
    ClosedByUserId       INT NULL,                  -- FK لجدول Users (يُضاف لاحقاً بقسم الصلاحيات)
    IsActive             BIT NOT NULL DEFAULT 1     -- الفترة الحالية المفتوحة للعمل
);

-- ==========================================
-- 5. أنواع المستندات (Document Types) - جدول مرجعي موحّد
-- بديل عن القائمة المنسدلة الهشة بالنظام القديم؛ يُستخدم بكل الجداول أدناه
-- ==========================================
CREATE TABLE DocumentTypes (
    DocumentTypeId       INT IDENTITY(1,1) PRIMARY KEY,
    Code                 NVARCHAR(30) NOT NULL UNIQUE,  -- SALES_CASH, SALES_CREDIT, SALES_RETURN_CASH, PURCHASE_CASH, PURCHASE_CREDIT,
                                                          -- PURCHASE_RETURN_CASH, PURCHASE_RETURN_CREDIT, QUOTATION, PURCHASE_ORDER,
                                                          -- CUSTOMS_CLEARANCE, STOCK_TRANSFER, STOCK_ISSUE, OPENING_BALANCE,
                                                          -- STOCK_ADJUST_INCREASE, STOCK_ADJUST_DECREASE, RECEIPT_VOUCHER,
                                                          -- PAYMENT_VOUCHER, JOURNAL_VOUCHER, JOURNAL_OPENING
    NameAr               NVARCHAR(100) NOT NULL,
    Category             TINYINT NOT NULL             -- 1=مستند تجاري (فاتورة/مردود/أمر), 2=سند مالي, 3=مستند مخزني
);

-- ==========================================
-- 6. رأس القيد المحاسبي (Journal Header) - القلب النابض للنظام
-- كل عملية مالية بالنظام (فاتورة، سند، تسوية) تُولّد سطراً هنا تلقائياً
-- ==========================================
CREATE TABLE JournalEntries (
    JournalEntryId       BIGINT IDENTITY(1,1) PRIMARY KEY,
    EntryNumber          INT NOT NULL,               -- رقم تسلسلي معروض للمستخدم (يُعاد الترقيم لكل سنة مالية)
    FiscalPeriodId       INT NOT NULL REFERENCES FiscalPeriods(FiscalPeriodId),
    EntryDate            DATE NOT NULL,
    DocumentTypeId       INT NOT NULL REFERENCES DocumentTypes(DocumentTypeId),
    SourceDocumentId     BIGINT NULL,                -- ربط بمصدر القيد (رقم الفاتورة/السند الذي ولّد هذا القيد) - Polymorphic
    Description          NVARCHAR(500) NULL,         -- البيان العام للقيد
    BranchId             INT NULL REFERENCES Branches(BranchId),
    CenterId             INT NULL REFERENCES Centers(CenterId),
    IsAutoGenerated       BIT NOT NULL DEFAULT 0,     -- مولّد تلقائياً من مستند، أم قيد يدوي مباشر
    CreatedByUserId      INT NULL,                   -- FK لاحقاً
    CreatedAt            DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    IsDeleted            BIT NOT NULL DEFAULT 0
);

CREATE UNIQUE INDEX IX_JournalEntries_Number ON JournalEntries(FiscalPeriodId, EntryNumber);

-- ==========================================
-- 7. تفاصيل القيد (Journal Lines) - سطور المدين/الدائن
-- ==========================================
CREATE TABLE JournalLines (
    JournalLineId         BIGINT IDENTITY(1,1) PRIMARY KEY,
    JournalEntryId        BIGINT NOT NULL REFERENCES JournalEntries(JournalEntryId),
    LineNumber            INT NOT NULL,               -- ترتيب السطر داخل القيد
    AccountId             INT NOT NULL REFERENCES Accounts(AccountId),
    Description           NVARCHAR(300) NULL,         -- بيان خاص بالسطر (يطغى على البيان العام لو موجود)
    CurrencyId            INT NOT NULL REFERENCES Currencies(CurrencyId),
    ExchangeRate          DECIMAL(18,6) NOT NULL DEFAULT 1,  -- التعادل وقت تسجيل القيد (Snapshot - لا يتأثر بتغيّر السعر لاحقاً)
    DebitAmount           DECIMAL(18,2) NOT NULL DEFAULT 0,
    CreditAmount          DECIMAL(18,2) NOT NULL DEFAULT 0,
    CenterId              INT NULL REFERENCES Centers(CenterId),
    CONSTRAINT CK_JournalLines_DebitOrCredit CHECK (
        (DebitAmount > 0 AND CreditAmount = 0) OR (CreditAmount > 0 AND DebitAmount = 0)
    )
);

CREATE INDEX IX_JournalLines_Account ON JournalLines(AccountId);
CREATE INDEX IX_JournalLines_Entry ON JournalLines(JournalEntryId);

القسم 3: الأصناف والمخزون (Items, Units, Pricing, Stock)
-- ==========================================
-- 1. الأصناف (بيانات الصنف الأساسية)
-- ==========================================
CREATE TABLE Items (
    ItemId               INT IDENTITY(1,1) PRIMARY KEY,
    ItemCode             INT NOT NULL UNIQUE,          -- رقم تسلسلي تلقائي (وليس إدخال يدوي كما اتفقنا)
    NameAr               NVARCHAR(200) NOT NULL,
    Barcode              NVARCHAR(50)  NULL,            -- باركود الوحدة الأساسية (باركود إضافي لكل وحدة بجدول منفصل أدناه)
    DepartmentId         INT NULL REFERENCES Departments(DepartmentId),
    ModelNo              NVARCHAR(100) NULL,            -- موديل
    LocationId           INT NULL REFERENCES StockLocations(LocationId),
    ImagePath            NVARCHAR(300) NULL,
    Specifications        NVARCHAR(1000) NULL,           -- المواصفات
    -- خيارات سلوكية:
    DecreaseQtyOnSale    BIT NOT NULL DEFAULT 1,        -- تناقص الكمية عند البيع (يُعطّل للخدمات مثلاً)
    PrintInKitchen       BIT NOT NULL DEFAULT 0,        -- طباعة بالمطبخ (قطاع مطاعم)
    -- الوحدات الافتراضية:
    IssueUnitId          INT NULL REFERENCES Units(UnitId),   -- وحدة الصرف
    ReceiveUnitId         INT NULL REFERENCES Units(UnitId),   -- وحدة التوريد
    -- تتبع الصلاحية والكمية:
    TrackExpiryDate      BIT NOT NULL DEFAULT 0,        -- اعتماد تاريخ الصلاحية
    LowQtyAlertLevel     DECIMAL(18,3) NULL,            -- تنبيه نقص كمية
    ExpiryAlertDays      INT NULL,                       -- تنبيه بفترة قبل الانتهاء (أيام)
    IsComposite          BIT NOT NULL DEFAULT 0,        -- صنف مركّب (له مكوّنات - BOM)
    IsActive             BIT NOT NULL DEFAULT 1,
    CreatedAt            DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    ModifiedAt           DATETIME2 NULL,
    IsDeleted            BIT NOT NULL DEFAULT 0
);

CREATE INDEX IX_Items_Department ON Items(DepartmentId);
CREATE INDEX IX_Items_Barcode ON Items(Barcode);
CREATE INDEX IX_Items_NameAr ON Items(NameAr);   -- لدعم البحث السريع بالكتابة (Autocomplete)

-- ==========================================
-- 2. وحدات الصنف (تعدد الوحدات لكل صنف + معامل تحويل)
-- ==========================================
CREATE TABLE ItemUnits (
    ItemUnitId           INT IDENTITY(1,1) PRIMARY KEY,
    ItemId               INT NOT NULL REFERENCES Items(ItemId),
    UnitId               INT NOT NULL REFERENCES Units(UnitId),
    ConversionFactor     DECIMAL(18,4) NOT NULL DEFAULT 1,  -- كم يساوي بالوحدة الأساسية (كرتون = 24 حبة)
    Barcode              NVARCHAR(50) NULL,                  -- باركود خاص بهذه الوحدة تحديداً
    IsBaseUnit           BIT NOT NULL DEFAULT 0,
    IsDeleted            BIT NOT NULL DEFAULT 0,
    UNIQUE(ItemId, UnitId)
);

-- ==========================================
-- 3. مكوّنات الصنف (BOM - للأصناف المركّبة/المصنّعة)
-- ==========================================
CREATE TABLE ItemComponents (
    ItemComponentId       INT IDENTITY(1,1) PRIMARY KEY,
    ParentItemId          INT NOT NULL REFERENCES Items(ItemId),      -- الصنف المركّب (النهائي)
    ComponentItemId       INT NOT NULL REFERENCES Items(ItemId),      -- الصنف المكوّن (الفرعي)
    UnitId                INT NOT NULL REFERENCES Units(UnitId),
    Quantity              DECIMAL(18,4) NOT NULL,
    IsDeleted             BIT NOT NULL DEFAULT 0
);

-- ==========================================
-- 4. شرائح التسعير (تعريف الشرائح: مستهلك/عميل/جملة/موزع/خاص)
-- ==========================================
CREATE TABLE PriceTiers (
    PriceTierId           INT IDENTITY(1,1) PRIMARY KEY,
    NameAr                NVARCHAR(50) NOT NULL,     -- مستهلك، عميل، جملة، موزع، خاص
    DisplayOrder          INT NOT NULL DEFAULT 0,
    IsDeleted             BIT NOT NULL DEFAULT 0
);

-- ==========================================
-- 5. أسعار الأصناف (مصفوفة: صنف × شريحة تسعير × عملة)
-- ==========================================
CREATE TABLE ItemPrices (
    ItemPriceId           INT IDENTITY(1,1) PRIMARY KEY,
    ItemId                INT NOT NULL REFERENCES Items(ItemId),
    PriceTierId           INT NOT NULL REFERENCES PriceTiers(PriceTierId),
    CurrencyId            INT NOT NULL REFERENCES Currencies(CurrencyId),
    SalePrice             DECIMAL(18,4) NOT NULL DEFAULT 0,
    ModifiedAt            DATETIME2 NULL,
    UNIQUE(ItemId, PriceTierId, CurrencyId)
);

-- ==========================================
-- 6. إحصائيات الصنف التلقائية (Cache محسوبة - لتسريع العرض بدل حساب حي كل مرة)
-- تُحدَّث تلقائياً بعد كل حركة عبر Trigger أو Application Service
-- ==========================================
CREATE TABLE ItemStatistics (
    ItemId                INT PRIMARY KEY REFERENCES Items(ItemId),
    LastPurchasePrice     DECIMAL(18,4) NULL,
    LowestPurchasePrice   DECIMAL(18,4) NULL,
    HighestPurchasePrice  DECIMAL(18,4) NULL,

القسم 6: الكاشير/POS — التفاصيل البرمجية

1. لماذا الكاشير يحتاج معمارية مختلفة قليلاً عن باقي الشاشات؟

الكاشير له طبيعة خاصة: يعمل باللمس، يحتاج استجابة فورية (لا تأخير شبكة محسوس)، ويدعم فواتير متعددة متوازية (1-4 كما رأينا بالصورة) — أي الكاشير يمكنه "تعليق" فاتورة والبدء بأخرى ثم العودة للأولى.

2. الكيان الإضافي: جلسة كاشير معلّقة (Held Orders)

-- جدول إضافي خاص بالكاشير فقط (لا يؤثر على TradeDocuments حتى تُرحَّل الفاتورة نهائياً)
CREATE TABLE POSHeldOrders (
    HeldOrderId          INT IDENTITY(1,1) PRIMARY KEY,
    SlotNumber           TINYINT NOT NULL,        -- 1, 2, 3, 4 (الفواتير المتوازية بالشاشة)
    UserId               INT NOT NULL REFERENCES Users(UserId),
    CashboxAccountId     INT NULL REFERENCES Accounts(AccountId),
    PriceTierId          INT NULL REFERENCES PriceTiers(PriceTierId),
    CreatedAt            DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    UNIQUE(SlotNumber, UserId)   -- كل مستخدم كاشير له 4 خانات مستقلة به فقط
);

CREATE TABLE POSHeldOrderLines (
    HeldOrderLineId       INT IDENTITY(1,1) PRIMARY KEY,
    HeldOrderId           INT NOT NULL REFERENCES POSHeldOrders(HeldOrderId),
    ItemId                INT NOT NULL REFERENCES Items(ItemId),
    UnitId                INT NOT NULL REFERENCES Units(UnitId),
    Quantity              DECIMAL(18,3) NOT NULL,
    UnitPrice             DECIMAL(18,4) NOT NULL
);

لاحظ: هذا منفصل تماماً عن TradeDocuments — فاتورة الكاشير المعلّقة ليست "فاتورة رسمية" بعد، فقط مسودة عمل. فقط عندما يضغط الكاشير "دفع/إنهاء"، تُحوَّل لسطر حقيقي بجدول TradeDocuments عبر نفس TradeDocumentService الذي بنيناه مسبقاً — إعادة استخدام كاملة للمنطق الموجود، الكاشير مجرد "واجهة سريعة" فوق نفس الخدمة.

// HisabTak.WPF/ViewModels/POS/CashierViewModel.cs
public partial class CashierViewModel : ObservableObject
{
    private readonly ITradeDocumentService _documentService;
    private readonly IPOSHeldOrderService _heldOrderService;
    private readonly IItemSearchService _itemSearch;

    [ObservableProperty] private int activeSlot = 1;   // الفاتورة النشطة حالياً (1-4)
    [ObservableProperty] private ObservableCollection<CartLineDto> cartLines = new();
    [ObservableProperty] private ObservableCollection<DepartmentButtonDto> departmentButtons = new();
    [ObservableProperty] private decimal totalAmount;

    // عند الضغط على زر صنف ملوّن بالشاشة
    [RelayCommand]
    private void AddItemToCart(ItemButtonDto item)
    {
        var existingLine = CartLines.FirstOrDefault(l => l.ItemId == item.ItemId);
        if (existingLine != null)
            existingLine.Quantity += 1;      // لو موجود، زد الكمية فقط
        else
            CartLines.Add(new CartLineDto { ItemId = item.ItemId, Quantity = 1, UnitPrice = item.CurrentPrice });

        RecalculateTotal();
        // حفظ فوري بجدول POSHeldOrderLines (كل ضغطة تُحفظ تلقائياً، لمنع فقدان البيانات لو انقطعت الكهرباء)
        _ = _heldOrderService.SaveDraftLineAsync(ActiveSlot, existingLine ?? CartLines.Last());
    }

    // التبديل بين الفواتير الأربع المتوازية
    [RelayCommand]
    private async Task SwitchSlotAsync(int slotNumber)
    {
        ActiveSlot = slotNumber;
        var savedOrder = await _heldOrderService.GetHeldOrderAsync(slotNumber);
        CartLines = new ObservableCollection<CartLineDto>(savedOrder?.Lines ?? new());
        RecalculateTotal();
    }

    // إتمام الدفع — هنا يحدث الربط الحقيقي بالنظام الأساسي
    [RelayCommand]
    private async Task CheckoutAsync(string paymentType)   // "cash" أو "credit"
    {
        var dto = new CreateTradeDocumentDto
        {
            DocumentTypeCode = paymentType == "cash" ? "SALES_CASH" : "SALES_CREDIT",
            Lines = CartLines.Select(l => new TradeDocumentLineDto { /* ... */ }).ToList()
        };

        // نفس الخدمة المستخدمة بفاتورة البيع العادية — لا كود مكرر إطلاقاً
        var documentId = await _documentService.SaveInvoiceAsync(dto);

        await _heldOrderService.ClearSlotAsync(ActiveSlot);   // تفريغ الخانة بعد إتمام البيع
        CartLines.Clear();

        // طباعة الفاتورة فوراً (تفصيلها بالقسم القادم)
        await _printService.PrintReceiptAsync(documentId, isSmallFormat: true);
    }
}

// HisabTak.Infrastructure/Backup/BackupService.cs
public class BackupService : IBackupService
{
    private readonly string _connectionString;

    public async Task<string> CreateBackupAsync(string backupFolderPath)
    {
        var dbName = GetDatabaseNameFromConnectionString(_connectionString);
        var fileName = $"{dbName}_{DateTime.Now:yyyyMMdd_HHmmss}.bak";
        var fullPath = Path.Combine(backupFolderPath, fileName);

        var sql = $@"BACKUP DATABASE [{dbName}] 
                     TO DISK = @BackupPath 
                     WITH FORMAT, INIT, NAME = 'HisabTak Full Backup';";

        using var connection = new SqlConnection(_connectionString);
        await connection.ExecuteAsync(sql, new { BackupPath = fullPath });

        return fullPath;
    }

    public async Task RestoreBackupAsync(string backupFilePath)
    {
        var dbName = GetDatabaseNameFromConnectionString(_connectionString);

        // يجب قطع كل الاتصالات النشطة بقاعدة البيانات أولاً قبل الاستعادة
        var killConnectionsSql = $@"
            ALTER DATABASE [{dbName}] SET SINGLE_USER WITH ROLLBACK IMMEDIATE;";

        var restoreSql = $@"
            RESTORE DATABASE [{dbName}] FROM DISK = @BackupPath WITH REPLACE;
            ALTER DATABASE [{dbName}] SET MULTI_USER;";

        using var connection = new SqlConnection("Server=.;Database=master;Trusted_Connection=True;");
        await connection.ExecuteAsync(killConnectionsSql);
        await connection.ExecuteAsync(restoreSql, new { BackupPath = backupFilePath });
    }
}

2. النسخ التلقائي المجدول (كما رأينا بالنظام القديم: "3 مسارات + تلقائي")

// HisabTak.WPF/Services/ScheduledBackupService.cs — يعمل بالخلفية طوال فترة تشغيل التطبيق
public class ScheduledBackupService : BackgroundService
{
    private readonly IBackupService _backupService;
    private readonly ISystemSettingsService _settings;

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            var settings = await _settings.GetBackupSettingsAsync();

            if (settings.AutoBackupEnabled && DateTime.Now.TimeOfDay >= settings.ScheduledBackupTime)
            {
                foreach (var path in settings.BackupPaths)  // يدعم 3 مسارات كالنظام القديم (فلاشة + قرص محلي + مجلد شبكة)
                {
                    try { await _backupService.CreateBackupAsync(path); }
                    catch (Exception ex) { _logger.LogError(ex, $"فشل النسخ الاحتياطي بالمسار: {path}"); }
                }
            }

            await Task.Delay(TimeSpan.FromHours(1), stoppingToken);  // فحص كل ساعة
        }
    }
}

// HisabTak.Infrastructure/Printing/Templates/InvoiceDocument.cs
public class InvoiceDocument : IDocument
{
    private readonly TradeDocumentPrintDto _invoice;
    private readonly bool _isSmallFormat;   // فاتورة كاشير صغيرة (80mm) أم فاتورة عادية A4

    public InvoiceDocument(TradeDocumentPrintDto invoice, bool isSmallFormat)
    {
        _invoice = invoice;
        _isSmallFormat = isSmallFormat;
    }

    public void Compose(IDocumentContainer container)
    {
        container.Page(page =>
        {
            page.Size(_isSmallFormat ? PageSizes.A6 : PageSizes.A4);
            page.Margin(1, Unit.Centimetre);

            page.Header().Element(ComposeHeader);
            page.Content().Element(ComposeContent);
            page.Footer().Element(ComposeFooter);
        });
    }

    private void ComposeHeader(IContainer container)
    {
        container.Row(row =>
        {
            row.RelativeItem().Column(col =>
            {
                col.Item().Text(_invoice.CompanyInfo.NameAr).FontSize(16).Bold();
                col.Item().Text(_invoice.CompanyInfo.AddressAr);
                col.Item().Text($"هاتف: {_invoice.CompanyInfo.Phone}");
            });

            if (!string.IsNullOrEmpty(_invoice.CompanyInfo.LogoPath))
                row.ConstantItem(80).Image(_invoice.CompanyInfo.LogoPath);
        });
    }

    private void ComposeContent(IContainer container)
    {
        container.Table(table =>
        {
            table.ColumnsDefinition(columns =>
            {
                columns.RelativeColumn(3);  // الصنف
                columns.RelativeColumn(1);  // الوحدة
                columns.RelativeColumn(1);  // الكمية
                columns.RelativeColumn(1);  // السعر
                columns.RelativeColumn(1);  // الإجمالي
            });

            table.Header(header =>
            {
                header.Cell().Text("الصنف").Bold();
                header.Cell().Text("الوحدة").Bold();
                header.Cell().Text("الكمية").Bold();
                header.Cell().Text("السعر").Bold();
                header.Cell().Text("الإجمالي").Bold();
            });

            foreach (var line in _invoice.Lines)
            {
                table.Cell().Text(line.ItemName);
                table.Cell().Text(line.UnitName);
                table.Cell().Text(line.Quantity.ToString("N2"));
                table.Cell().Text(line.UnitPrice.ToString("N2"));
                table.Cell().Text(line.LineTotal.ToString("N2"));
            }
        });
    }

    private void ComposeFooter(IContainer container)
    {
        container.Column(col =>
        {
            col.Item().AlignRight().Text($"الإجمالي: {_invoice.NetTotal:N2} {_invoice.CurrencyName}").Bold();
            if (!string.IsNullOrEmpty(_invoice.CompanyInfo.FooterNote))
                col.Item().AlignCenter().Text(_invoice.CompanyInfo.FooterNote).FontSize(9);
        });
    }
}

public class ExcelImportService : IExcelImportService
{
    public async Task<ImportResultDto> ImportItemsFromExcelAsync(Stream fileStream, ImportOptionsDto options)
    {
        using var workbook = new XLWorkbook(fileStream);
        var sheet = workbook.Worksheet(1);
        var result = new ImportResultDto();

        foreach (var row in sheet.RowsUsed().Skip(1))   // تخطي صف الرأس
        {
            var itemName = row.Cell(2).GetString();

            // خيار "تنبيه بتكرار الاسم" من النظام القديم
            if (options.WarnOnDuplicateName && await _itemService.ExistsByNameAsync(itemName))
            {
                result.Warnings.Add($"الصنف '{itemName}' موجود مسبقاً بالسطر {row.RowNumber()}");
                if (!options.OverwriteDuplicates) continue;
            }

            var newItem = await _itemService.CreateItemAsync(new CreateItemDto
            {
                NameAr = itemName,
                Barcode = row.Cell(3).GetString(),
                SalePrice = row.Cell(5).GetValue<decimal>()
            });

            // خيار "إضافة الأصناف للرصيد الافتتاحي" من النظام القديم
            if (options.AddToOpeningBalance)
            {
                await _stockService.CreateOpeningBalanceEntryAsync(newItem.ItemId, row.Cell(6).GetValue<decimal>());
            }

            result.SuccessCount++;
        }

        return result;
    }

النتيجة: الفاتورة تُحفظ بنجاح، تظهر بالتقارير، لكن الصندوق لا يزيد فعلياً! هذا خطأ صامت وخطير جداً بنظام محاسبي — لن تكتشفه إلا بعد أسابيع حين تلاحظ أن أرصدة الصناديق لا تتطابق مع الواقع، وبحلول ذلك الوقت قد تكون مئات الفواتير الخاطئة قد سُجّلت.

مع Unit Tests (الحل)

أنت (أو أي مبرمج) كتبت مسبقاً اختبار بسيط جداً يقول:

"لو أنشأت فاتورة بيع نقدي بقيمة 500 ريال، يجب أن يكون بالقيد الناتج سطر واحد يزيد الصندوق بـ 500 بالضبط، وسطر آخر يزيد المبيعات بـ 500 بالضبط."

الآن، بعد أي تعديل بالكود (سواء اليوم أو بعد 5 سنوات)، قبل أن يُنشر البرنامج للعميل، يُشغَّل هذا الاختبار تلقائياً خلال أقل من ثانية. لو الكود المعدَّل حذف سطر "زيادة الصندوق" بالخطأ، الاختبار يفشل فوراً ويظهر لك رسالة حمراء واضحة:

❌ FAILED: BuildJournalEntry_CreatesBalancedEntry_ForCashSale
   Expected: الصندوق = 500
   Actual: الصندوق = 0


فتكتشف الخطأ قبل أن يصل لأي عميل حقيقي إطلاقاً، وتصلحه فوراً وأنت لا زلت تعمل على نفس السطر بالضبط. هذا الفرق الجوهري: اكتشاف الخطأ خلال ثانية أثناء البرمجة، بدل اكتشافه بعد أسابيع من الاستخدام الفعلي بأموال حقيقية.

خلاصة بسيطة: Unit Tests = حراس آليون صغار، كل واحد منهم يراقب "قاعدة" واحدة بسيطة (مثل: "القيد يجب أن يكون متوازناً دائماً")، ويصرخ فوراً لو أي تعديل مستقبلي كسر هذه القاعدة، بدلاً من انتظار اكتشافها يدوياً بالصدفة.

تسجيل باقي القرارات وتأكيدها

كل قراراتك (النسخ الاحتياطي المحلي الاختياري، سكربت Batch للقالب الجاهز، ورفض الدعم عن بعد حالياً) محفوظة بالفعل بذاكرتي بدقة. الآن أعتبر التحليل والتصميم الكامل منتهياً بالكامل بكل الأقسام العشرة.
