    public DateTime? ModifiedAt { get; protected set; }

📁 HisabTak.Domain/Entities/Accounting/JournalEntry.cs

using HisabTak.Domain.Common;
using HisabTak.Domain.Exceptions;

namespace HisabTak.Domain.Entities.Accounting;

public class JournalEntry : BaseEntity
{
    public long JournalEntryId { get; private set; }
    public int EntryNumber { get; private set; }
    public int FiscalPeriodId { get; private set; }
    public DateTime EntryDate { get; private set; }
    public int DocumentTypeId { get; private set; }
    public long? SourceDocumentId { get; private set; }
    public string? Description { get; private set; }

    private readonly List<JournalLine> _lines = new();
    public IReadOnlyCollection<JournalLine> Lines => _lines.AsReadOnly();

    // منطق العمل الأساسي يعيش هنا داخل الـ Domain، وليس متروكاً للواجهة
    public void AddLine(JournalLine line)
    {
        _lines.Add(line);
    }

    public void ValidateBalance()
    {
        var totalDebit = _lines.Sum(l => l.DebitAmount);
        var totalCredit = _lines.Sum(l => l.CreditAmount);

        if (totalDebit != totalCredit)
            throw new UnbalancedJournalEntryException(totalDebit, totalCredit);
    }
}

📁 HisabTak.Application/Accounting/Services/JournalService.cs

using HisabTak.Application.Common.Interfaces;
using HisabTak.Domain.Entities.Accounting;

namespace HisabTak.Application.Accounting.Services;

public class JournalService : IJournalService
{
    private readonly IUnitOfWork _unitOfWork;
    private readonly ICurrentUserService _currentUser;

    public JournalService(IUnitOfWork unitOfWork, ICurrentUserService currentUser)
    {
        _unitOfWork = unitOfWork;
        _currentUser = currentUser;
    }

    public async Task<long> CreateJournalEntryAsync(CreateJournalEntryDto dto)
    {
        var entry = JournalEntry.Create(dto.EntryDate, dto.DocumentTypeId, dto.Description);

        foreach (var lineDto in dto.Lines)
        {
            entry.AddLine(JournalLine.Create(
                lineDto.AccountId, lineDto.DebitAmount, lineDto.CreditAmount, lineDto.CurrencyId));
        }

        entry.ValidateBalance();   // يرمي استثناء تلقائياً لو غير متوازن — الواجهة فقط تعرض رسالة الخطأ

        await _unitOfWork.JournalEntries.AddAsync(entry);
        await _unitOfWork.SaveChangesAsync();

        return entry.JournalEntryId;
    }
}

Views و Stored Procedures للتقارير الختامية

1. الرصيد الحالي لكل حساب (View أساسي يُبنى عليه كل شيء)

CREATE OR ALTER VIEW vw_AccountBalances AS
SELECT
    a.AccountId,
    a.AccountCode,
    a.NameAr,
    a.MainGroupId,
    a.Nature,
    a.CurrencyId,
    a.OpeningBalance * (CASE WHEN a.OpeningBalanceNature = 1 THEN 1 ELSE -1 END)
        + ISNULL(SUM(jl.DebitAmount - jl.CreditAmount), 0) AS NetBalance
        -- NetBalance موجب = مدين، سالب = دائن (نظام إشارة موحّد يبسّط كل الحسابات اللاحقة)
FROM Accounts a
LEFT JOIN JournalLines jl ON jl.AccountId = a.AccountId
LEFT JOIN JournalEntries je ON je.JournalEntryId = jl.JournalEntryId AND je.IsDeleted = 0
WHERE a.IsDeleted = 0
GROUP BY a.AccountId, a.AccountCode, a.NameAr, a.MainGroupId, a.Nature, a.CurrencyId, 
         a.OpeningBalance, a.OpeningBalanceNature;

2. Stored Procedure: حساب المتاجرة (Trading Account)

CREATE OR ALTER PROCEDURE sp_GetTradingAccountReport
    @FromDate DATE,
    @ToDate DATE,
    @CurrencyId INT,
    @BranchId INT = NULL
AS
BEGIN
    SET NOCOUNT ON;

    -- مخزون أول المدة: نجمع كل حركات المخزون (بالتكلفة) قبل @FromDate
    DECLARE @OpeningStockValue DECIMAL(18,2) =
        (SELECT ISNULL(SUM(CASE WHEN sm.Direction = 1 THEN sm.Quantity * sm.UnitCost
                                  ELSE -sm.Quantity * sm.UnitCost END), 0)
         FROM StockMovements sm
         WHERE sm.MovementDate < @FromDate AND sm.IsDeleted = 0);

    -- إجمالي المشتريات خلال الفترة (فواتير الشراء نقد+آجل)
    DECLARE @TotalPurchases DECIMAL(18,2) =
        (SELECT ISNULL(SUM(td.NetTotal), 0)
         FROM TradeDocuments td
         INNER JOIN DocumentTypes dt ON dt.DocumentTypeId = td.DocumentTypeId
         WHERE dt.Code IN ('PURCHASE_CASH','PURCHASE_CREDIT')
           AND td.DocumentDate BETWEEN @FromDate AND @ToDate
           AND td.CurrencyId = @CurrencyId
           AND td.Status = 1 AND td.IsDeleted = 0
           AND (@BranchId IS NULL OR td.BranchId = @BranchId));

    -- إجمالي مردودات المشتريات
    DECLARE @PurchaseReturns DECIMAL(18,2) =
        (SELECT ISNULL(SUM(td.NetTotal), 0)
         FROM TradeDocuments td
         INNER JOIN DocumentTypes dt ON dt.DocumentTypeId = td.DocumentTypeId
         WHERE dt.Code IN ('PURCHASE_RETURN_CASH','PURCHASE_RETURN_CREDIT')
           AND td.DocumentDate BETWEEN @FromDate AND @ToDate
           AND td.CurrencyId = @CurrencyId
           AND td.Status = 1 AND td.IsDeleted = 0
           AND (@BranchId IS NULL OR td.BranchId = @BranchId));

    DECLARE @NetPurchases DECIMAL(18,2) = @TotalPurchases - @PurchaseReturns;

    -- مخزون آخر المدة: قيمة أرصدة المخزون الحالية بتاريخ @ToDate
    DECLARE @ClosingStockValue DECIMAL(18,2) =
        (SELECT ISNULL(SUM(sb.Quantity * ist.AverageCost), 0)
         FROM StockBalances sb
         INNER JOIN ItemStatistics ist ON ist.ItemId = sb.ItemId);

    DECLARE @COGS DECIMAL(18,2) = @OpeningStockValue + @NetPurchases - @ClosingStockValue;
                                    -- تكلفة البضاعة المباعة

    -- إجمالي المبيعات وصافيها
    DECLARE @TotalSales DECIMAL(18,2) =
        (SELECT ISNULL(SUM(td.NetTotal), 0)
         FROM TradeDocuments td
         INNER JOIN DocumentTypes dt ON dt.DocumentTypeId = td.DocumentTypeId
         WHERE dt.Code IN ('SALES_CASH','SALES_CREDIT')
           AND td.DocumentDate BETWEEN @FromDate AND @ToDate
           AND td.CurrencyId = @CurrencyId
           AND td.Status = 1 AND td.IsDeleted = 0
           AND (@BranchId IS NULL OR td.BranchId = @BranchId));

    DECLARE @SalesReturns DECIMAL(18,2) =
        (SELECT ISNULL(SUM(td.NetTotal), 0)
         FROM TradeDocuments td
         INNER JOIN DocumentTypes dt ON dt.DocumentTypeId = td.DocumentTypeId
         WHERE dt.Code IN ('SALES_RETURN_CASH','SALES_RETURN_CREDIT')
           AND td.DocumentDate BETWEEN @FromDate AND @ToDate
           AND td.CurrencyId = @CurrencyId
           AND td.Status = 1 AND td.IsDeleted = 0
           AND (@BranchId IS NULL OR td.BranchId = @BranchId));

    DECLARE @NetSales DECIMAL(18,2) = @TotalSales - @SalesReturns;
    DECLARE @GrossProfit DECIMAL(18,2) = @NetSales - @COGS;

    -- إرجاع كل الأرقام بصف واحد نظيف
    SELECT
        @OpeningStockValue  AS OpeningStockValue,
        @TotalPurchases     AS TotalPurchases,
        @PurchaseReturns    AS PurchaseReturns,
        @NetPurchases       AS NetPurchases,
        @ClosingStockValue  AS ClosingStockValue,
        @COGS               AS CostOfGoodsSold,
        @TotalSales         AS TotalSales,
        @SalesReturns       AS SalesReturns,
        @NetSales           AS NetSales,
        @GrossProfit        AS GrossProfitOrLoss;
END

3. Stored Procedure: الأرباح والخسائر (يعتمد على المتاجرة)

CREATE OR ALTER PROCEDURE sp_GetProfitAndLossReport
    @FromDate DATE,
    @ToDate DATE,
    @CurrencyId INT,
    @BranchId INT = NULL
AS
BEGIN
    SET NOCOUNT ON;

    -- إجمالي الربح/الخسارة من المتاجرة (نستدعي نفس منطق المتاجرة كدالة مساعدة أو نكرر الحساب هنا)
    DECLARE @GrossProfit DECIMAL(18,2);
    -- (بالتطبيق الفعلي: استخراج @GrossProfit من نتيجة sp_GetTradingAccountReport عبر جدول مؤقت)

    -- الإيرادات الثانوية (كل حسابات مجموعة "الإيرادات" ما عدا المبيعات نفسها)
    SELECT @TotalOtherRevenue = ISNULL(SUM(ab.NetBalance) * -1, 0)   -- الإيرادات دائنة، نقلبها لموجب
    FROM vw_AccountBalances ab
    INNER JOIN AccountMainGroups mg ON mg.MainGroupId = ab.MainGroupId
    WHERE mg.Code LIKE '4%'   -- مجموعة الإيرادات (استثناء حساب المبيعات الرئيسي لو مطلوب تفصيل أدق)
      AND ab.CurrencyId = @CurrencyId;

    -- المصروفات (كل حسابات مجموعة "المصروفات" مفصّلة حساب حساب)
    SELECT
        a.AccountId,
        a.NameAr,
        ab.NetBalance AS ExpenseAmount
    INTO #Expenses
    FROM vw_AccountBalances ab
    INNER JOIN Accounts a ON a.AccountId = ab.AccountId
    INNER JOIN AccountMainGroups mg ON mg.MainGroupId = ab.MainGroupId
    WHERE mg.Code LIKE '3%'   -- مجموعة المصروفات
      AND ab.CurrencyId = @CurrencyId
      AND ab.NetBalance <> 0;

    DECLARE @TotalExpenses DECIMAL(18,2) = (SELECT ISNULL(SUM(ExpenseAmount), 0) FROM #Expenses);
    DECLARE @NetProfit DECIMAL(18,2) = @GrossProfit + @TotalOtherRevenue - @TotalExpenses;

    -- النتيجة: ملخص + تفصيل المصروفات
    SELECT @GrossProfit AS GrossProfit, @TotalOtherRevenue AS OtherRevenue,
           @TotalExpenses AS TotalExpenses, @NetProfit AS NetProfitOrLoss;

    SELECT * FROM #Expenses ORDER BY ExpenseAmount DESC;

    DROP TABLE #Expenses;
END

CREATE OR ALTER PROCEDURE sp_GetBalanceSheetReport
    @AsOfDate DATE,
    @CurrencyId INT
AS
BEGIN
    SET NOCOUNT ON;
    -- الميزانية = كل حسابات الأصول والالتزامات وحقوق الملكية (مجموعات 1 و 2) بأرصدتها حتى تاريخ معين
    SELECT a.AccountCode, a.NameAr, mg.Code AS MainGroupCode, mg.NameAr AS MainGroupName,
           ab.NetBalance
    FROM vw_AccountBalances ab
    INNER JOIN Accounts a ON a.AccountId = ab.AccountId
    INNER JOIN AccountMainGroups mg ON mg.MainGroupId = ab.MainGroupId
    WHERE (mg.Code LIKE '1%' OR mg.Code LIKE '2%')
      AND ab.CurrencyId = @CurrencyId
      AND ab.NetBalance <> 0
    ORDER BY mg.Code, a.AccountCode;
END
GO

CREATE OR ALTER PROCEDURE sp_GetTrialBalanceReport
    @AsOfDate DATE,
    @CurrencyId INT
AS
BEGIN
    SET NOCOUNT ON;
    -- ميزان المراجعة = كل الحسابات بلا استثناء (كل المجموعات) بأرصدتها
    SELECT a.AccountCode, a.NameAr, mg.Code AS MainGroupCode,
           CASE WHEN ab.NetBalance > 0 THEN ab.NetBalance ELSE 0 END AS DebitBalance,
           CASE WHEN ab.NetBalance < 0 THEN ABS(ab.NetBalance) ELSE 0 END AS CreditBalance
    FROM vw_AccountBalances ab
    INNER JOIN Accounts a ON a.AccountId = ab.AccountId
    INNER JOIN AccountMainGroups mg ON mg.MainGroupId = ab.MainGroupId
    WHERE ab.CurrencyId = @CurrencyId AND ab.NetBalance <> 0
    ORDER BY mg.Code, a.AccountCode;
END

خدمة الترحيل التلقائي للقيود (Auto-Posting Service)

هذه الخدمة هي "القلب الخفي" للنظام — كل عملية مالية (فاتورة، سند، تسوية) يجب أن تولّد قيداً محاسبياً صحيحاً ومتوازناً تلقائياً، دون أن يكتب المستخدم القيد يدوياً أبداً.

1. التصميم: نمط "استراتيجية الترحيل" (Posting Strategy Pattern)

بما أن كل نوع مستند له قواعد ترحيل مختلفة (فاتورة بيع نقدي تخصم من الصندوق وتُضيف للمبيعات، بينما فاتورة بيع آجل تُضيف للعميل بدل الصندوق)، نصمم واجهة موحّدة ولكل نوع مستند "استراتيجية" منفصلة تنفّذها:

// HisabTak.Application/Accounting/Interfaces/IPostingStrategy.cs
namespace HisabTak.Application.Accounting.Interfaces;

public interface IPostingStrategy
{
    string DocumentTypeCode { get; }   // 'SALES_CASH', 'PURCHASE_CREDIT'...
    Task<JournalEntry> BuildJournalEntryAsync(object sourceDocument);
}

// HisabTak.Application/Accounting/Services/PostingStrategies/SalesCashPostingStrategy.cs
namespace HisabTak.Application.Accounting.Services.PostingStrategies;

public class SalesCashPostingStrategy : IPostingStrategy
{
    public string DocumentTypeCode => "SALES_CASH";

    private readonly IAccountLookupService _accountLookup;

    public SalesCashPostingStrategy(IAccountLookupService accountLookup)
    {
        _accountLookup = accountLookup;
    }

    public async Task<JournalEntry> BuildJournalEntryAsync(object sourceDocument)
    {
        var invoice = (TradeDocument)sourceDocument;

        var entry = JournalEntry.Create(
            entryDate: invoice.DocumentDate,
            documentTypeId: invoice.DocumentTypeId,
            sourceDocumentId: invoice.TradeDocumentId,
            description: $"فاتورة بيع نقدي رقم {invoice.DocumentNumber}");

        // مدين: الصندوق (الخزينة المختارة بالفاتورة)
        entry.AddLine(JournalLine.Create(
            accountId: invoice.CashboxAccountId!.Value,
            debit: invoice.NetTotal, credit: 0,
            currencyId: invoice.CurrencyId, exchangeRate: invoice.ExchangeRate));

        // دائن: حساب المبيعات (يُحدَّد تلقائياً من إعدادات النظام أو من القسم)
        var salesAccountId = await _accountLookup.GetSalesRevenueAccountAsync();
        entry.AddLine(JournalLine.Create(
            accountId: salesAccountId,
            debit: 0, credit: invoice.NetTotal,
            currencyId: invoice.CurrencyId, exchangeRate: invoice.ExchangeRate));

        // لو فيه ضريبة، سطر إضافي لحساب "الضريبة المستحقة"
        if (invoice.TaxValue > 0)
        {
            var taxAccountId = await _accountLookup.GetTaxPayableAccountAsync();
            // تعديل حساب المبيعات ليكون بدون الضريبة + سطر ضريبة منفصل
            entry.AddLine(JournalLine.Create(
                accountId: taxAccountId, debit: 0, credit: invoice.TaxValue,
                currencyId: invoice.CurrencyId, exchangeRate: invoice.ExchangeRate));
        }

        entry.ValidateBalance();   // يتأكد أن مدين = دائن قبل الحفظ (يرمي استثناء لو لا)
        return entry;
    }
}

// مثال ثانٍ: فاتورة بيع آجل — نفس الواجهة، منطق مختلف
public class SalesCreditPostingStrategy : IPostingStrategy
{

    {
        var invoice = (TradeDocument)sourceDocument;
        var entry = JournalEntry.Create(invoice.DocumentDate, invoice.DocumentTypeId,
            invoice.TradeDocumentId, $"فاتورة بيع آجل رقم {invoice.DocumentNumber}");

        entry.AddLine(JournalLine.Create(invoice.AccountId!.Value, invoice.NetTotal, 0,
            invoice.CurrencyId, invoice.ExchangeRate));

        var salesAccountId = await _accountLookup.GetSalesRevenueAccountAsync();
        entry.AddLine(JournalLine.Create(salesAccountId, 0, invoice.NetTotal,
            invoice.CurrencyId, invoice.ExchangeRate));

        entry.ValidateBalance();
        return entry;
    }
}

2. المُنسّق المركزي (Posting Coordinator) — يختار الاستراتيجية الصحيحة تلقائياً

// HisabTak.Application/Accounting/Services/AutoPostingService.cs
namespace HisabTak.Application.Accounting.Services;

public class AutoPostingService : IAutoPostingService
{
    private readonly IEnumerable<IPostingStrategy> _strategies;
    private readonly IUnitOfWork _unitOfWork;

    // كل استراتيجيات الترحيل تُحقن تلقائياً هنا عبر DI (Dependency Injection)
    public AutoPostingService(IEnumerable<IPostingStrategy> strategies, IUnitOfWork unitOfWork)
    {
        _strategies = strategies;
        _unitOfWork = unitOfWork;
    }

    public async Task<long> PostDocumentAsync(string documentTypeCode, object sourceDocument)
    {
        var strategy = _strategies.FirstOrDefault(s => s.DocumentTypeCode == documentTypeCode)
            ?? throw new PostingStrategyNotFoundException(documentTypeCode);

        var journalEntry = await strategy.BuildJournalEntryAsync(sourceDocument);

        await _unitOfWork.JournalEntries.AddAsync(journalEntry);
        await _unitOfWork.SaveChangesAsync();

        return journalEntry.JournalEntryId;
    }
}

{

    private readonly IStockMovementService _stockService;
    private readonly IUnitOfWork _unitOfWork;

    public async Task<long> SaveInvoiceAsync(CreateTradeDocumentDto dto)
    {
        // 1. إنشاء الفاتورة نفسها
        var document = TradeDocument.Create(dto);
        await _unitOfWork.TradeDocuments.AddAsync(document);
        await _unitOfWork.SaveChangesAsync();   // نحفظ أولاً للحصول على TradeDocumentId

        // 2. تحديث المخزون تلقائياً (خدمة منفصلة، تفصيلها بالأسفل)
        await _stockService.ApplyStockMovementForDocumentAsync(document);

        // 3. الترحيل المحاسبي التلقائي — سطر واحد فقط يُشغّل كل المنطق أعلاه
        var journalEntryId = await _postingService.PostDocumentAsync(dto.DocumentTypeCode, document);

        // 4. ربط الفاتورة بالقيد المولّد
        document.LinkJournalEntry(journalEntryId);
        await _unitOfWork.SaveChangesAsync();

        return document.TradeDocumentId;
    }
}

بعكس الترحيل المحاسبي (الذي يحتاج منطقاً مختلفاً جذرياً لكل نوع)، حركة المخزون فعلياً نمطان فقط: وارد أو منصرف. لذا نستخدم قاموساً بسيطاً بدل كلاسات متعددة:

// HisabTak.Application/Inventory/Services/StockDirectionResolver.cs
public static class StockDirectionResolver
{
    private static readonly Dictionary<string, StockDirection> _map = new()
    {
        ["PURCHASE_CASH"]           = StockDirection.In,
        ["PURCHASE_CREDIT"]         = StockDirection.In,
        ["SALES_RETURN_CASH"]       = StockDirection.In,
        ["SALES_RETURN_CREDIT"]     = StockDirection.In,
        ["STOCK_ADJUST_INCREASE"]   = StockDirection.In,
        ["OPENING_BALANCE"]         = StockDirection.In,

        ["SALES_CASH"]              = StockDirection.Out,
        ["SALES_CREDIT"]            = StockDirection.Out,
        ["PURCHASE_RETURN_CASH"]    = StockDirection.Out,
        ["PURCHASE_RETURN_CREDIT"]  = StockDirection.Out,
        ["STOCK_ADJUST_DECREASE"]   = StockDirection.Out,
        ["STOCK_ISSUE"]             = StockDirection.Out,   // سند صرف مخزني

        // عروض الأسعار وأوامر الشراء لا تُحرّك المخزون إطلاقاً
        ["QUOTATION"]               = StockDirection.None,
        ["PURCHASE_ORDER"]          = StockDirection.None,
    };

    public static StockDirection Resolve(string documentTypeCode) =>
        _map.TryGetValue(documentTypeCode, out var dir) ? dir : StockDirection.None;
}

public enum StockDirection { None, In, Out }

نقطة تصميمية مهمة: QUOTATION (عرض السعر) وPURCHASE_ORDER (أمر الشراء) مسجّلان بـ StockDirection.None — أي لا تُحرّك المخزون إطلاقاً، لأنهما مستندات "نية" وليسا عمليات فعلية. هذا يطبّق منطقاً صحيحاً كان يجب مراعاته بدقة (عرض السعر ليس بيعاً فعلياً).

3. التحويل المخزني حالة خاصة (Direction مزدوج: منصرف من مصدر + وارد لهدف)

// HisabTak.Application/Inventory/Services/StockMovementService.cs
public class StockMovementService : IStockMovementService
{
    private readonly IUnitOfWork _unitOfWork;
    private readonly ICostingService _costingService;

    public async Task ApplyStockMovementForDocumentAsync(TradeDocument document)
    {
        var documentTypeCode = document.DocumentType.Code;

        // حالة خاصة: التحويل المخزني (وارد ومنصرف بنفس الوقت لمستودعين مختلفين)
        if (documentTypeCode == "STOCK_TRANSFER")
        {
            await ApplyTransferAsync(document);
            return;
        }

        var direction = StockDirectionResolver.Resolve(documentTypeCode);
        if (direction == StockDirection.None) return;   // عروض أسعار وأوامر شراء: لا شيء يحدث

        foreach (var line in document.Lines)
        {
            // 1. تحويل الكمية للوحدة الأساسية دائماً (مهما كانت وحدة السطر: كرتون، حبة...)
            var baseQty = await ConvertToBaseUnitAsync(line.ItemId, line.UnitId, line.Quantity);

            // 2. حساب تكلفة الوحدة وقت هذه الحركة (فقط للحركات الواردة، تُستخدم لاحقاً بحساب المتوسط)
            decimal? unitCost = direction == StockDirection.In
                ? await _costingService.CalculateUnitCostAsync(line, document)
                : null;

            // 3. تسجيل الحركة بالسجل التاريخي (لا يُعدَّل أبداً لاحقاً)
            await _unitOfWork.StockMovements.AddAsync(StockMovement.Create(
                itemId: line.ItemId, warehouseId: line.WarehouseId,
                movementDate: document.DocumentDate,
                documentTypeId: document.DocumentTypeId,
                sourceDocumentId: document.TradeDocumentId,
                direction: direction, quantity: baseQty,
                unitCost: unitCost, expiryDate: line.ExpiryDate));

            // 4. تحديث الرصيد الحي (Cache) — بزيادة أو نقصان حسب الاتجاه
            await UpdateStockBalanceAsync(line.ItemId, line.WarehouseId, line.ExpiryDate,
                direction == StockDirection.In ? baseQty : -baseQty);

            // 5. تحديث إحصائيات الصنف (آخر سعر، متوسط تكلفة...)
            await _costingService.UpdateItemStatisticsAsync(line, document, direction);
        }
    }

    private async Task ApplyTransferAsync(TradeDocument document)
    {
        foreach (var line in document.Lines)
        {
            var baseQty = await ConvertToBaseUnitAsync(line.ItemId, line.UnitId, line.Quantity);

            // منصرف من المستودع المصدر
            await UpdateStockBalanceAsync(line.ItemId, document.FromWarehouseId!.Value, line.ExpiryDate, -baseQty);
            await _unitOfWork.StockMovements.AddAsync(StockMovement.Create(
                line.ItemId, document.FromWarehouseId.Value, document.DocumentDate,
                document.DocumentTypeId, document.TradeDocumentId, StockDirection.Out, baseQty, null, line.ExpiryDate));

            // وارد للمستودع الهدف (بنفس التكلفة المنقولة، وليس تكلفة جديدة)
            await UpdateStockBalanceAsync(line.ItemId, document.ToWarehouseId!.Value, line.ExpiryDate, baseQty);
            await _unitOfWork.StockMovements.AddAsync(StockMovement.Create(
                line.ItemId, document.ToWarehouseId.Value, document.DocumentDate,
                document.DocumentTypeId, document.TradeDocumentId, StockDirection.In, baseQty, line.CostAtTime, line.ExpiryDate));
        }
    }

    private async Task<decimal> ConvertToBaseUnitAsync(int itemId, int unitId, decimal quantity)
    {
        var itemUnit = await _unitOfWork.ItemUnits.GetByItemAndUnitAsync(itemId, unitId);
        return quantity * itemUnit.ConversionFactor;   // مثال: 2 كرتون × 24 = 48 حبة
    }

    private async Task UpdateStockBalanceAsync(int itemId, int warehouseId, DateTime? expiryDate, decimal deltaQty)
    {
        var balance = await _unitOfWork.StockBalances.GetOrCreateAsync(itemId, warehouseId, expiryDate);
        balance.AdjustQuantity(deltaQty);   // Domain method: يمنع الرصيد من أن يصبح سالباً لو أردنا هذا القيد لاحقاً
    }
}

4. خدمة التكلفة (طريقتان يدعمهما إعداد النظام: آخر شراء أو متوسط تكلفة)

// HisabTak.Application/Inventory/Services/CostingService.cs
public class CostingService : ICostingService
{
    private readonly ISystemSettingsService _settings;

    public async Task<decimal> CalculateUnitCostAsync(TradeDocumentLine line, TradeDocument document)
    {
        // للحركات الواردة (شراء)، التكلفة = سعر الشراء الفعلي بالفاتورة نفسها
        return line.NetUnitPrice;
    }

    public async Task UpdateItemStatisticsAsync(TradeDocumentLine line, TradeDocument document, StockDirection direction)
    {
        var stats = await _unitOfWork.ItemStatistics.GetOrCreateAsync(line.ItemId);
        var costingMethod = await _settings.GetCostingMethodAsync();   // من CompanyInfo.CostingMethod

        if (direction == StockDirection.In)
        {
            stats.UpdateLastPurchase(line.NetUnitPrice, line.Quantity);

            if (costingMethod == CostingMethod.AverageCost)
            {
                // المتوسط المرجّح: (الرصيد القديم × تكلفته + الوارد الجديد × تكلفته) ÷ إجمالي الكمية
                var currentBalance = await _unitOfWork.StockBalances.GetTotalQuantityAsync(line.ItemId);
                var newAverage = CalculateWeightedAverage(
                    oldQty: currentBalance, oldCost: stats.AverageCost ?? 0,
                    newQty: line.Quantity, newCost: line.NetUnitPrice);
                stats.UpdateAverageCost(newAverage);
            }
            else // LastPurchasePrice
            {
                stats.UpdateAverageCost(line.NetUnitPrice);
            }
        }
        else // Out (بيع)
        {
            stats.UpdateLastSale(line.NetUnitPrice, line.Quantity);
        }

        await _unitOfWork.SaveChangesAsync();
    }

    private decimal CalculateWeightedAverage(decimal oldQty, decimal oldCost, decimal newQty, decimal newCost)
    {
        var totalQty = oldQty + newQty;
        if (totalQty == 0) return newCost;
        return ((oldQty * oldCost) + (newQty * newCost)) / totalQty;
    }
}

5. نقطة أمان مهمة: منع البيع بكمية أكبر من المتوفر (اختياري حسب إعداد)

// داخل Domain Entity: StockBalance.cs
public class StockBalance : BaseEntity
{
    public decimal Quantity { get; private set; }

    public void AdjustQuantity(decimal delta, bool allowNegative = false)
    {
        var newQuantity = Quantity + delta;

        if (!allowNegative && newQuantity < 0)
            throw new InsufficientStockException(ItemId, Math.Abs(delta), Quantity);

        Quantity = newQuantity;
    }
}

بحيث عند محاولة بيع كمية أكبر من الرصيد الفعلي، يُرمى استثناء واضح قبل حفظ الفاتورة أصلاً (وليس بعد الحفظ بمشكلة رصيد سالب مكتشفة لاحقاً كما قد يحدث بأنظمة أضعف).

6. الفائدة الكبرى لهذا التصميم — نقطة تكامل واحدة نظيفة

لاحظ من الكود بالرسالة السابقة:

await _stockService.ApplyStockMovementForDocumentAsync(document);   // سطر واحد فقط
await _postingService.PostDocumentAsync(dto.DocumentTypeCode, document);  // سطر واحد فقط

كل الفواتير السبعة (بيع، شراء، مردودات، تحويل، صرف مخزني) تمرّ بنفس هاتين الخطوتين بالضبط من TradeDocumentService — لا تكرار كود، ولا احتمال نسيان تحديث المخزون بشاشة معينة كما قد يحدث لو كان الكود منسوخاً بـ7 أماكن منفصلة.

هل هذا واضح ومقبول؟ بهذا نكون أنجزنا العمود الفقري الكامل للنظام (Schema + هيكلة + التقارير الختامية + الترحيل المحاسبي التلقائي + تحديث المخزون التلقائي).
